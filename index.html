var API_KEY = "ใส่ API key ของคุณ"; var SHEET_ID = "ใส่ Spreadsheet ID ของคุณ"; function doGet(e) { var callback = e.parameter.callback; var action = e.parameter.action || "ask"; var result; if (action === "tts") { result = handleTTS(e.parameter.text || ""); } else { result = handleAsk(e.parameter.question || ""); } var json = JSON.stringify(result); return ContentService .createTextOutput(callback + "(" + json + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } function handleAsk(question) { try { var context = readSheetContext(); var prompt = "คุณคือผู้ช่วย AI ประจำสวน\n" + "ข้อมูลสวนปัจจุบัน:\n" + context + "\n\n" + "ตอบคำถามภาษาที่ผู้ใช้ใช้ (ไทย/อังกฤษ/ญี่ปุ่น)\n" + "ตอบกระชับ ไม่เกิน 4 ประโยค\n\n" + "คำถาม: " + question; var url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=" + API_KEY; var payload = { contents: [{ parts: [{ text: prompt }] }] }; var options = { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true }; var response = UrlFetchApp.fetch(url, options); var data = JSON.parse(response.getContentText()); var answer = ""; if (data.candidates && data.candidates[0] && data.candidates[0].content) { answer = data.candidates[0].content.parts[0].text; } else { answer = "ขอโทษครับ ไม่สามารถประมวลผลได้"; } return { answer: answer }; } catch (err) { return { error: "Apps Script ERROR: " + err.message }; } } /* ===== Thai TTS: server เรียก Google Translate TTS แทน browser ===== */ function handleTTS(text) { try { text = String(text).substring(0, 200); if (!text) return { error: "no text" }; var url = "https://translate.google.com/translate_tts" + "?ie=UTF-8&client=tw-ob&tl=th&q=" + encodeURIComponent(text); var response = UrlFetchApp.fetch(url, { headers: { "User-Agent": "Mozilla/5.0" }, muteHttpExceptions: true }); if (response.getResponseCode() !== 200) { return { error: "TTS HTTP " + response.getResponseCode() }; } var bytes = response.getBlob().getBytes(); var base64 = Utilities.base64Encode(bytes); return { audio: base64 }; } catch (err) { return { error: "TTS ERROR: " + err.message }; } } function readSheetContext() { try { var sheet = SpreadsheetApp.openById(SHEET_ID).getSheets()[0]; var values = sheet.getDataRange().getValues(); var lines = []; for (var i = 0; i < values.length && i < 50; i++) { lines.push(values[i].join(" | ")); } return lines.join("\n"); } catch (err) { return "(ไม่สามารถอ่านข้อมูลสวนได้: " + err.message + ")"; } }
